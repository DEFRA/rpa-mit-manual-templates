Private Sub BK_Click()

    dBody = "BK"

    ActiveWorkbook.Worksheets("Main Page - " & WBTYpe).Range("B3").Value = strOrg & "  -  " & dBody
    ActiveWorkbook.Worksheets("Main Page - " & WBTYpe).Range("B4").Value = "Bee Keeping Scheme"
    
    Dim response As VbMsgBoxResult
    response = MsgBox("Bee Keeping has been selected" & vbNewLine & "Is this correct?", vbYesNo)
    If response = vbYes Then
        Call SelectFormST.Hide
        Call SetValRng
        Call PrepAPorAR
    End If

End Sub

Private Sub CloseWorkbook_Click()
    Dim response As VbMsgBoxResult
    response = MsgBox("Are you sure you want to close this workbook?", vbYesNo)
    If response = vbYes Then
        Application.DisplayAlerts = False
        ActiveWorkbook.Close
        Application.DisplayAlerts = True
        Application.ScreenUpdating = True
    End If
End Sub

Private Sub UserForm_Activate()
    Me.Left = GetLeftFormOffset(Me.Width)
    Me.Top = GetTopFormOffset(Me.Height)
End Sub
