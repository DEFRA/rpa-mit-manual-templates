


Private Sub LS_Click()

    dBody = "LS"    
    dBody = strOrg & dBody      ' #### e.g.NELS     

End Sub

Private Sub CS_Click()

    
    If strOrg = "RDT" Then
        dBody = "CP"        'the new 2014 - 2020 Pillar 2 Programme
    ElseIf strOrg = "EA" Then
        dBody = "EA"
    Else
        dBody = "CS"
    End If    
 
    If strOrg <> "EA" Then
        dBody = strOrg & dBody 'e.g. NECS
    End If
    
End Sub

Private Sub P1_Click()
    
    dBody = "P1"    'NE P1 schemes   
  
    
End Sub
